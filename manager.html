async function deleteSelected() { const checks = document.querySelectorAll('.del-check:checked'); if (checks.length === 0) return; if (!confirm(checks.length + '件のデータを削除しますか？')) return; // 重要：行番号が大きい順（後ろから）に並び替えることで行ズレを防ぐ const rowsToDelete = Array.from(checks).map(c => parseInt(c.value)).sort((a, b) => b - a); for (let row of rowsToDelete) { await fetch(GAS_URL, { method: 'POST', mode: 'no-cors', body: JSON.stringify({ action: 'delete', row: row }) }); } alert('削除が完了しました'); load(); }
