// goToUserInfo 関数内を以下のように修正（上記 Code.gs に合わせる） async function goToUserInfo() { const checked = document.querySelectorAll('.row-checkbox:checked'); if (checked.length === 0) return alert("ユーザーを選択してください"); const names = Array.from(checked).map(cb => cb.getAttribute('data-name')); document.querySelectorAll('.view').forEach(v => v.classList.remove('active')); document.getElementById('view-info').classList.add('active'); const res = await fetch(GAS_URL + '?names=' + encodeURIComponent(names.join(','))); const data = await res.json(); const tbody = document.querySelector('#record-table tbody'); tbody.innerHTML = ''; (data.records || []).forEach(r => { tbody.innerHTML += `${r.name}${r.出勤}${r.休憩入り}${r.休憩戻り}${r.退勤}`; }); }
